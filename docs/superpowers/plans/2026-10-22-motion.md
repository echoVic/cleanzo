# Main-Module Motion — Implementation Plan

> **For agentic workers:** In this project the tasks are executed by the Codex CLI, several tasks per run (see "Execution batches"). **Executors do not commit**: the reviewer checks and commits each task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Every main module gets the transitions and feedback specified in the approved motion design (one signature moment when cleaning completes, continuity for everything that moves, a reduced-motion path that keeps fades). Flows, buttons and copy stay the same apart from the listed feedback changes.

**Architecture:**
- A few motion primitives live in `src/ui/motion/`:
  - `motion.ts` with levels and durations;
  - `Collapse`, `usePresence`/`Leave` and `useFlip`;
  - `NumberTicker`, `useIndicator`, `useAnimatedHeight`, and the `fly*` helpers.
- The motion tokens and keyframes go in `src/style.css`.
- Pages adopt these primitives without changing their state machines. Everything is CSS keyframes/transitions plus the Web Animations API: no new dependency.
- When Web Animations is unavailable (bun tests, static rendering), the primitives do nothing and commit synchronously. Existing tests therefore keep their assertions.
- A dev-only mock of the mygo runtime (`?mock`) lets the reviewer drive the real pages in a browser.

**Tech Stack:** TypeScript 7 + React 19 + Radix (dialog, tabs, checkbox) + Tailwind v4 + Vite 8, bun test. No Go changes.

**Spec:** `docs/superpowers/specs/2026-10-22-motion-design.md` (Chinese). Section numbers below refer to it. The approved prototype `docs/superpowers/specs/2026-10-22-motion-prototype.html` shows every motion in a browser and is the visual reference. Where the prototype and the spec differ, the spec wins.

## Global Constraints

- **No new dependencies.** Do not use the View Transitions API or `@starting-style`. Do not run commands that need the network.
- **Flows, buttons and copy do not change**, except the feedback changes in spec §七. New strings go through `t()` with both `zh` and `en` entries in `src/i18n.ts`.
- **Durations and easing** come from the CSS tokens (`--t-press` 90ms, `--t-fade` 120ms, `--t-state` 180ms, `--t-layout` 240ms, `--t-move` 360ms, `--e-out`, `--e-move`, `--e-in`) or from `DUR`/`EASE` in `src/ui/motion/motion.ts`. Literal durations are allowed only for these named spec exceptions:
  - the 140/100ms row menu;
  - the 150ms dialog close;
  - the 180ms overlay fade;
  - the 450ms scan-bar hold;
  - the clean-complete moment (540/290/440/420ms);
  - the 900ms highlight fade.
- **Reduced motion** follows spec §4.3:
  - Movement and scaling are removed. Opacity and colour changes stay, and so do spinners.
  - Never leave a Radix exit animation at 0ms duration; use the fade keyframes instead.
- **Without Web Animations** (`motionLevel() === "none"`) every primitive is a synchronous no-op. Existing test assertions must not change. Add every new test file to the `test:web` script in `package.json`.
- **`src/shared.ts`** must keep working with the fake element from `tests/setup-dom.ts`. That element has only `textContent`, `disabled`, `dataset`, `style` and `addEventListener`; it has no `classList`, `className` or `replaceChildren`.
- **Do not edit `src/mygo.ts`**, any Go file, `交付/`, `build/`, `dist/` or `outputs/`. Do not edit the AI tools, skills or settings pages: they only inherit the shared components.
- **Style:**
  - TypeScript: terse Chinese comments like the existing code. Every new export gets a one-line doc comment.
  - CSS: new classes use the `cz-` prefix. New rules are not wrapped in `@layer`, which matches the existing `style.css`. `tests/css-tokens.test.ts` must stay green.
- **After each task** these must pass: `bun run typecheck`, `bun run test:web`, and `bun run build:web -- --outDir "$TMPDIR/cleanzo-motion-web"`. Write build output to TMPDIR, never to `dist/`.

```ts
class MockChannel<V> implements AsyncIterable<V> {
  onmessage: ((value: V) => void) | null;
  onclose: (() => void) | null = null;
  closed = false;
  private queue: V[] = [];
  private waiters: ((r: IteratorResult<V>) => void)[] = [];
  constructor(onmessage?: (value: V) => void) { this.onmessage = onmessage ?? null; }
  /** 后端侧发送；页面已关闭通道时返回 false，流应停止。 */
  send(value: V): boolean {
    if (this.closed) return false;
    if (this.onmessage) { this.onmessage(value); return true; }
    const w = this.waiters.shift();
    if (w) w({ value, done: false }); else this.queue.push(value);
    return true;
  }
  close(): void {
    if (this.closed) return;
    this.closed = true;
    this.waiters.splice(0).forEach((w) => w({ value: undefined as never, done: true }));
    this.onclose?.();
  }
  [Symbol.asyncIterator](): AsyncIterator<V> {
    return {
      next: () => {
        if (this.queue.length) return Promise.resolve({ value: this.queue.shift()!, done: false });
        if (this.closed) return Promise.resolve({ value: undefined as never, done: true });
        return new Promise((r) => this.waiters.push(r));
      },
      return: () => { this.close(); return Promise.resolve({ value: undefined as never, done: true }); },
    };
  }
}
```

```ts
// 动效基础：时长、曲线、当前能不能动。时长与 style.css 的 --t-* 令牌一致（设计第四节）。
export const DUR = { press: 90, fade: 120, state: 180, layout: 240, move: 360, tick: 420, spark: 450 } as const;
export type DurKey = keyof typeof DUR;

export const EASE = {
  out: "cubic-bezier(0.16, 1, 0.3, 1)",
  move: "cubic-bezier(0.2, 0, 0, 1)",
  in: "cubic-bezier(0.3, 0, 0.8, 0.15)",
} as const;

/** full 完整动效；reduced 只留淡入淡出；none 没有 Web Animations（bun 测试、静态渲染），一切直接到位。 */
export type MotionLevel = "full" | "reduced" | "none";

let forced: MotionLevel | undefined;

/** 测试用：固定动效级别；传 undefined 恢复自动判断。 */
export function setMotionLevelForTest(level: MotionLevel | undefined): void {
  forced = level;
}

/** 当前动效级别。 */
export function motionLevel(): MotionLevel {
  if (forced) return forced;
  if (typeof window === "undefined" || typeof Element === "undefined" || typeof Element.prototype.animate !== "function") return "none";
  if (typeof window.matchMedia === "function" && window.matchMedia("(prefers-reduced-motion: reduce)").matches) return "reduced";
  return "full";
}

/** 这一档在当前级别下的实际时长（ms）：reduced 只保留 fade，none 一律为 0。 */
export function dur(key: DurKey): number {
  const level = motionLevel();
  if (level === "none") return 0;
  if (level === "reduced") return key === "fade" ? DUR.fade : 0;
  return DUR[key];
}

/** 播放一段 Web Animation；时长为 0 或环境不支持时立即完成。 */
export function play(
  el: Element | null | undefined,
  keyframes: Keyframe[],
  ms: number,
  easing: string = EASE.out,
  opts: KeyframeAnimationOptions = {},
): Promise<void> {
  if (!el || !(ms > 0) || typeof el.animate !== "function") return Promise.resolve();
  const a = el.animate(keyframes, { duration: ms, easing, fill: "backwards", ...opts });
  return a.finished.then(() => undefined, () => undefined);
}

/** 平滑滚动；减弱动态效果或不支持时直接跳过去。 */
export function scrollBehavior(): ScrollBehavior {
  return motionLevel() === "full" ? "smooth" : "auto";
}
```

```tsx
import * as React from "react";
import { EASE, dur, motionLevel, play } from "./motion";

/** 列表里的一项：leaving 表示正在离场，只为播放收起动画而保留。 */
export interface Presence<T> {
  item: T;
  key: string;
  leaving: boolean;
}

type Leaving<T> = ReadonlyMap<string, { item: T; index: number }>;

/** 把离场项按原来的位置插回当前列表（纯函数，便于单测）。 */
export function mergePresence<T>(items: readonly T[], keyOf: (item: T) => string, leaving: Leaving<T>): Presence<T>[] {
  const out: Presence<T>[] = items.map((item) => ({ item, key: keyOf(item), leaving: false }));
  if (leaving.size === 0) return out;
  const present = new Set(out.map((p) => p.key));
  [...leaving.entries()]
    .filter(([key]) => !present.has(key))
    .sort((a, b) => a[1].index - b[1].index)
    .forEach(([key, v]) => out.splice(Math.min(v.index, out.length), 0, { item: v.item, key, leaving: true }));
  return out;
}

/**
 * 列表项离场：从 items 里消失的项再保留一个离场时长（leaving=true），之后移除。
 * enabled=false 的那次更新（搜索过滤、整表重置）直接移除；none 级别从不保留，测试里同步。
 */
export function usePresence<T>(items: readonly T[], keyOf: (item: T) => string, enabled = true): Presence<T>[] {
  const [leaving, setLeaving] = React.useState<Leaving<T>>(() => new Map());
  const prev = React.useRef<readonly T[]>(items);
  const timers = React.useRef<number[]>([]);
  React.useEffect(() => () => timers.current.forEach((id) => window.clearTimeout(id)), []);
  React.useLayoutEffect(() => {
    const before = prev.current;
    prev.current = items;
    if (before === items) return;
    const now = new Set(items.map(keyOf));
    // 重新出现的项不再离场。
    setLeaving((m) => {
      if (![...m.keys()].some((k) => now.has(k))) return m;
      const next = new Map(m);
      now.forEach((k) => next.delete(k));
      return next;
    });
    if (!enabled || motionLevel() === "none") return;
    const gone: [string, { item: T; index: number }][] = [];
    before.forEach((item, index) => {
      const key = keyOf(item);
      if (!now.has(key)) gone.push([key, { item, index }]);
    });
    if (gone.length === 0) return;
    setLeaving((m) => new Map([...m, ...gone]));
    const keys = gone.map(([k]) => k);
    timers.current.push(window.setTimeout(() => {
      setLeaving((m) => {
        const next = new Map(m);
        keys.forEach((k) => next.delete(k));
        return next;
      });
    }, Math.max(dur("layout"), dur("fade")) + 40));
  });
  return mergePresence(items, keyOf, leaving);
}

/** 行收起：高度、内边距、边框归零并淡出；父容器有 gap 时用负外边距抵掉，移除时不再跳一下。 */
export function collapseOut(el: HTMLElement | null): void {
  if (!el) return;
  el.style.pointerEvents = "none";
  const ms = dur("layout");
  if (!(ms > 0)) {
    void play(el, [{ opacity: 1 }, { opacity: 0 }], dur("fade"), "linear", { fill: "forwards" });
    return;
  }
  const cs = getComputedStyle(el);
  const gap = el.parentElement ? parseFloat(getComputedStyle(el.parentElement).rowGap) || 0 : 0;
  const last = !el.nextElementSibling;
  const height = el.getBoundingClientRect().height;
  el.style.overflow = "hidden";
  void play(el, [{ opacity: 1 }, { opacity: 0 }], dur("state"), EASE.in, { fill: "forwards" });
  void play(el, [
    { height: `${height}px`, paddingTop: cs.paddingTop, paddingBottom: cs.paddingBottom, borderTopWidth: cs.borderTopWidth, borderBottomWidth: cs.borderBottomWidth, marginTop: cs.marginTop, marginBottom: cs.marginBottom },
    { height: "0px", paddingTop: "0px", paddingBottom: "0px", borderTopWidth: "0px", borderBottomWidth: "0px", marginTop: last && gap ? `${-gap}px` : cs.marginTop, marginBottom: !last && gap ? `${-gap}px` : cs.marginBottom },
  ], ms, EASE.move, { fill: "forwards" });
}

/** 网格项离场：缩小淡出（减弱时只淡出）。 */
export function popOut(el: HTMLElement | null): void {
  if (!el) return;
  el.style.pointerEvents = "none";
  if (dur("state") > 0) void play(el, [{ opacity: 1, transform: "none" }, { opacity: 0, transform: "scale(0.92)" }], dur("state"), EASE.in, { fill: "forwards" });
  else void play(el, [{ opacity: 1 }, { opacity: 0 }], dur("fade"), "linear", { fill: "forwards" });
}

/** leaving 变为 true 时播放离场动画：collapse 用于列表行，pop 用于网格项。 */
export function useLeaveAnimation(ref: React.RefObject<HTMLElement | null>, leaving: boolean, kind: "collapse" | "pop" = "collapse"): void {
  React.useLayoutEffect(() => {
    if (!leaving) return;
    if (kind === "pop") popOut(ref.current);
    else collapseOut(ref.current);
  }, [leaving, kind, ref]);
}

/** 给唯一的宿主子元素挂上离场动画，不加包裹层（不影响 `.a + .a` 相邻选择器）。 */
export function Leave({ leaving, kind = "collapse", children }: {
  leaving: boolean;
  kind?: "collapse" | "pop";
  children: React.ReactElement;
}) {
  const ref = React.useRef<HTMLElement | null>(null);
  useLeaveAnimation(ref, leaving, kind);
  return React.cloneElement(children as React.ReactElement<Record<string, unknown>>, {
    ref,
    "aria-hidden": leaving || undefined,
    inert: leaving || undefined,
  });
}
```

- [ ] **Step 5: `flip.ts`.** The old positions are read during render, while the DOM still shows the previous commit. That keeps the motion continuous even when it is interrupted mid-flight.

```ts
import * as React from "react";
import { EASE, dur, motionLevel } from "./motion";

type Pos = { x: number; y: number };

function measure(box: HTMLElement): Map<string, Pos> {
  const base = box.getBoundingClientRect();
  const out = new Map<string, Pos>();
  box.querySelectorAll<HTMLElement>(":scope > [data-key]").forEach((el) => {
    const r = el.getBoundingClientRect();
    out.set(el.dataset.key ?? "", { x: r.left - base.left, y: r.top - base.top });
  });
  return out;
}

/**
 * 列表换位：keys 顺序变化时，容器的直接子元素（[data-key]）从旧位置滑到新位置。
 * 只传留在列表里的 key；离场项由 usePresence 收起，不参与换位。enabled=false 的那次更新不播放。
 */
export function useFlip(container: React.RefObject<HTMLElement | null>, keys: readonly string[], enabled = true, max = 40): void {
  const sig = keys.join("\u0000");
  const committed = React.useRef(sig);
  const before = React.useRef<Map<string, Pos> | null>(null);
  if (sig !== committed.current && enabled && container.current && motionLevel() === "full") {
    before.current = measure(container.current);
  }
  React.useLayoutEffect(() => {
    if (sig === committed.current) return;
    committed.current = sig;
    const first = before.current;
    before.current = null;
    const box = container.current;
    if (!first || !box || !(dur("move") > 0)) return;
    const els = box.querySelectorAll<HTMLElement>(":scope > [data-key]");
    els.forEach((el) => el.getAnimations().forEach((a) => { if (a.id === "cz-flip") a.cancel(); }));
    const last = measure(box);
    let n = 0;
    els.forEach((el) => {
      const key = el.dataset.key ?? "";
      const p = first.get(key);
      const q = last.get(key);
      if (!p || !q || n >= max) return;
      const dx = p.x - q.x;
      const dy = p.y - q.y;
      if (Math.abs(dx) < 0.5 && Math.abs(dy) < 0.5) return;
      n++;
      const a = el.animate([{ transform: `translate(${dx}px, ${dy}px)` }, { transform: "none" }], { duration: dur("move"), easing: EASE.move });
      a.id = "cz-flip";
    });
  }, [sig, container, max]);
}
```

```ts
/** 内容变化时外框高度平滑过渡：观察 inner 的尺寸，变化时让 outer 从旧高度过渡到新高度。 */
export function useAnimatedHeight(outer: React.RefObject<HTMLElement | null>, inner: React.RefObject<HTMLElement | null>, enabled: boolean): void {
  React.useLayoutEffect(() => {
    const box = outer.current;
    const body = inner.current;
    if (!enabled || !box || !body || typeof ResizeObserver === "undefined") return;
    let last = box.getBoundingClientRect().height;
    const ro = new ResizeObserver(() => {
      const next = box.getBoundingClientRect().height;
      const from = last;
      last = next;
      if (Math.abs(next - from) < 1 || !(dur("layout") > 0)) return;
      box.style.overflow = "hidden";
      void play(box, [{ height: `${from}px` }, { height: `${next}px` }], dur("layout"), EASE.move).then(() => { box.style.overflow = ""; });
    });
    ro.observe(body);
    return () => ro.disconnect();
  }, [enabled, outer, inner]);
}
```

```tsx
/**
 * 展开收起：外层 grid 行高 0fr ↔ 1fr 过渡，内容淡入淡出；收起动画结束后卸载内容。
 * 过渡期间才裁剪溢出，静止展开时不裁剪，行菜单和焦点环不会被切掉。
 */
export function Collapse({ open, children, className, appear = false, animateResize = false }: CollapseProps) {
  const [mounted, setMounted] = React.useState(open);
  const [expanded, setExpanded] = React.useState(open && !appear);
  const [moving, setMoving] = React.useState(false);
  const outer = React.useRef<HTMLDivElement>(null);
  const inner = React.useRef<HTMLDivElement>(null);
  useAnimatedHeight(outer, inner, animateResize && expanded && !moving);

  React.useLayoutEffect(() => {
    const ms = Math.max(dur("layout"), dur("fade"));
    if (open) {
      setMounted(true);
      if (expanded) return;
      if (!(ms > 0) || motionLevel() === "none") { setExpanded(true); return; }
      setMoving(true);
      // 先以收起态挂上，下一帧再展开，过渡才会发生。
      const raf = requestAnimationFrame(() => setExpanded(true));
      const timer = window.setTimeout(() => setMoving(false), ms + 60);
      return () => { cancelAnimationFrame(raf); window.clearTimeout(timer); };
    }
    setExpanded(false);
    if (!(ms > 0)) { setMounted(false); setMoving(false); return; }
    setMoving(true);
    const timer = window.setTimeout(() => { setMounted(false); setMoving(false); }, ms + 60);
    return () => window.clearTimeout(timer);
    // 只响应 open 的变化
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [open]);

  if (!mounted && !open) return null;
  return (
    <div ref={outer} className={cn("cz-collapse", className)} data-state={expanded ? "open" : "closed"} data-moving={moving ? "" : undefined}>
      <div ref={inner} className="cz-collapse-in">{children}</div>
    </div>
  );
}
```

```tsx
export interface NumberTickerProps {
  value: number;
  format: (n: number) => string;
  /** 首次挂载时从这个值滚到 value（例如扫描开始时从 0 累加）；不传则直接显示 value。 */

/** 数字滚动：数值变化时在 dur("tick") 内滚到新值，中途可改目标；减弱动态效果时直接显示。 */
export function NumberTicker({ value, format, from, as = "span", className }: NumberTickerProps) {
  const ref = React.useRef<HTMLElement | null>(null);
  const shown = React.useRef(from ?? value);
  const fmt = React.useRef(format);
  fmt.current = format;
  const [initial] = React.useState(() => format(from ?? value));
  React.useLayoutEffect(() => {
    const el = ref.current;
    if (!el) return;
    const node = el.firstChild ?? el.appendChild(document.createTextNode(""));
    const start = shown.current;
    const ms = dur("tick");
    if (!(ms > 0) || start === value || !Number.isFinite(start)) {
      shown.current = value;
      node.nodeValue = fmt.current(value);
      return;
    }
    let raf = 0;
    const t0 = performance.now();
    const step = (now: number) => {
      const p = Math.min(1, (now - t0) / ms);
      shown.current = p < 1 ? start + (value - start) * (1 - Math.pow(1 - p, 4)) : value;
      node.nodeValue = fmt.current(shown.current);
      if (p < 1) raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
    return () => cancelAnimationFrame(raf);
  }, [value]);
  return React.createElement(as, { ref, className }, initial);
}
```

```ts
  React.useLayoutEffect(() => {
    const box = container.current;
    if (!box) return;
    const place = (animate: boolean) => {
      const el = box.querySelector<HTMLElement>(selector);
      if (!el || !el.offsetWidth) return;
      setStyle({ width: el.offsetWidth, transform: `translateX(${el.offsetLeft}px)`, transition: animate ? undefined : "none" });

    place(false);
    const mo = new MutationObserver(() => place(true));
    mo.observe(box, { subtree: true, attributes: true, attributeFilter: ["data-state", "aria-selected"] });
    const ro = typeof ResizeObserver === "undefined" ? null : new ResizeObserver(() => place(false));
    ro?.observe(box);
    return () => { mo.disconnect(); ro?.disconnect(); };
  }, [container, selector]);
  return style;
```

```ts
// fly.ts
import { EASE, dur, play } from "./motion";

/** 页面坐标里的矩形（DOMRect 的子集，测试里不依赖 DOM）。 */
export interface Box { left: number; top: number; width: number; height: number }

/** 图片按 object-fit: contain 放进 stage 后实际占的矩形。 */
export function containRect(stage: Box, width: number, height: number): Box {
  const ratio = width / height;
  let w = stage.width;
  let h = w / ratio;
  if (h > stage.height) { h = stage.height; w = h * ratio; }
  return { left: stage.left + (stage.width - w) / 2, top: stage.top + (stage.height - h) / 2, width: w, height: h };
}

/** 文字共享元素：替身从起点飞到 to 的位置并换成 to 的颜色，期间隐藏 to。 */
export function flyText(from: { rect: Box; fontSize: number; color: string }, to: HTMLElement): Promise<void> {
  const ms = dur("move");
  const r1 = to.getBoundingClientRect();
  if (!(ms > 0) || !r1.width) return Promise.resolve();
  const cs = getComputedStyle(to);
  const ghost = document.createElement("div");
  ghost.className = "cz-fly";
```

```ts
export function mergeRefs<T>(...refs: (React.Ref<T> | undefined)[]): React.RefCallback<T> {
  return (node) => {
    for (const ref of refs) {
      if (typeof ref === "function") ref(node);
      else if (ref) (ref as React.MutableRefObject<T | null>).current = node;
    }
  };
```

```ts
export function isSparkShift(prev: readonly number[], next: readonly number[]): boolean {
```
