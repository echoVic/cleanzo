# 「应用」页：更新、占用、退出和卸载 — 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 侧栏「应用卸载」改成「应用」：一个 App 列表，行上显示有没有新版本，点开一行能更新、看占用并清缓存、退出、卸载（macOS）。

**Architecture:** Go 侧新增 `appupdates.go`（跨平台：类型、版本比较、URL 校验、受限下载、三个绑定）和 `appupdates_darwin.go`（四个版本来源、调度、Homebrew 升级、清缓存），非 macOS 走 `appupdates_other.go` 的空实现。前端把「已安装」标签换成 `AppsList`：筛选条 + 行内展开；残留仍用现有 `LeftoversTab`，挂在「残留」筛选下。

**Tech Stack:** Go 1.27（标准库 `net/http`、`encoding/xml`、`encoding/json`），mygo 0.7.1 绑定和 Channel，React 19 + Tailwind v4，bun test。

**Spec:** `docs/superpowers/specs/2026-11-02-apps-hub-design.md`（先读它，计划按它写）。

## Global Constraints

- 只请求 `https://`；主机是 `localhost`、`*.localhost`、回环、私有、未指定或链路本地地址的 URL，以及空 URL，一律不发请求，状态 `unknown`、原因 `bad-url`。重定向每一跳都过同样的校验，最多 5 次。
- 每个请求超时 10 秒，响应体最多读 2 MB（`2 << 20`），超出算 `error`/`bad-feed`。同时最多 6 个请求。User-Agent 是 `Cleanzo`，不带 Cookie。
- 版本只拿 App 包里的 `CFBundleShortVersionString`（Sparkle 另用 `CFBundleVersion`）和来源比；不用 brew 记的 installed 版本，不用 `brew outdated`。
- 比不出大小时状态是 `unknown`/`uncomparable`，绝不算「有更新」。
- Cleanzo 只执行 `brew upgrade --cask <token>`，不下载、不替换任何 App 包。前端只传 App 的 id，动作参数（token、trackId、URL）只在 Go 侧。
- 只在「应用」页开着时联网；没有后台检查、通知、红点、批量更新。
- Windows：没有更新、残留、清缓存、增长这几项，筛选条不显示；行展开只有占用（App 大小）和卸载。
- 所有新文案中英文都要有；Windows 下字典值不能出现 `tests/i18n-windows.test.ts` 禁止的词（废纸篓/Trash 会被词表换成回收站/Recycle Bin，可以用）。
- 注释、提交信息沿用仓库风格：Go 注释英文，前端注释中文。执行者不提交；评审者（Claude）在沙箱外复跑测试后提交。

package main

import (
	"context"
	"errors"
	"fmt"
	"io"
	"net"
	"net/http"
	"net/url"
	"strconv"
	"strings"
	"sync"
	"time"

// AppUpdate is one installed app's update state, streamed by
// CheckAppUpdates. The page shows Current → Latest and an Update button for
// State "update"; the button sends only ID, and the action stays in Go.
type AppUpdate struct {
	ID       string `json:"id"`       // AppInfo.ID; "__done" ends the stream
	Source   string `json:"source"`   // "mas" | "brew" | "sparkle" | "electron" | ""
	Current  string `json:"current"`  // the bundle's version
	Latest   string `json:"latest"`   // the source's newest version
	State    string `json:"state"`    // "update" | "current" | "unknown" | "error"
	Reason   string `json:"reason"`   // why unknown or error: no-source, bad-url, network, rate-limited, bad-feed, uncomparable, not-found
	NotesURL string `json:"notesUrl"` // release notes; https only
	Action   string `json:"action"`   // "brew" | "open-store" | "open-app" | ""
}

// AppUpgradeResult is what UpgradeApp did.
type AppUpgradeResult struct {
	ID      string `json:"id"`
	Done    bool   `json:"done"`    // brew: the bundle now has Latest; open-*: opened
	Version string `json:"version"` // the bundle's version afterwards (brew)
	Code    string `json:"code"`    // "in-use" | "brew-failed" | "not-checked" | "unsupported" | "busy" | ""
	Detail  string `json:"detail"`  // brew's last output line
	Command string `json:"command"` // the command to run in Terminal when brew failed
}

// appUpdateAction is what the Update button does for an app. It is only
// ever built in Go from what a check found.
type appUpdateAction struct {
	kind   string // "brew" | "open-store" | "open-app"
	token  string // brew cask token
	path   string // the app bundle
	url    string // macappstore:// link
	latest string // what the check found, to confirm a brew upgrade
}

// ---- versions ---------------------------------------------------------------

// cleanVersion drops a leading v and everything from the first separator
// that ends the dotted part: "1.171.0 (89666)", "1.171.0,89666",
// "7.0.0_260812_01" and "3.0.0-beta.2" all keep their leading numbers.
func cleanVersion(s string) string {
	s = strings.TrimSpace(s)
	s = strings.TrimPrefix(strings.TrimPrefix(s, "v"), "V")
	if i := strings.IndexAny(s, "-+_,( "); i >= 0 {
		s = s[:i]
	}
	return s
}

// versionParts splits a cleaned version into numbers; nil when any part is
// not a plain number, so it cannot be compared.
func versionParts(s string) []int {
	if s == "" {
		return nil
	}
	fields := strings.Split(s, ".")
	parts := make([]int, 0, len(fields))
	for _, f := range fields {
		n, err := strconv.Atoi(f)
		if f == "" || err != nil || n < 0 {
			return nil
		}
		parts = append(parts, n)
	}
	return parts
}

// updateState compares the bundle's version with the source's. Versions
// that cannot be compared are never reported as an update.
func updateState(current, latest string) (state, reason string) {
	c, l := versionParts(cleanVersion(current)), versionParts(cleanVersion(latest))
	if c == nil || l == nil {
		return "unknown", "uncomparable"
	}
	if compareVersions(l, c) > 0 {
		return "update", ""
	}
	return "current", ""
}

// ---- fetching feeds -----------------------------------------------------------

const (
	appFeedLimit     = 2 << 20
	appFeedTimeout   = 10 * time.Second
	appFeedRedirects = 5
)

var (
	errFeedURL      = errors.New("bad-url")
	errFeedRate     = errors.New("rate-limited")
	errFeedBad      = errors.New("bad-feed")
	errFeedNotFound = errors.New("not-found")
)

// feedURLOK reports whether a feed may be fetched: https, a public host
// name or address, no credentials.
func feedURLOK(raw string) bool {
	u, err := url.Parse(strings.TrimSpace(raw))
	if err != nil || u.Scheme != "https" || u.User != nil || u.Host == "" {
		return false
	}
	host := strings.ToLower(u.Hostname())
	if host == "" || host == "localhost" || strings.HasSuffix(host, ".localhost") {
		return false
	}
	if ip := net.ParseIP(host); ip != nil {
		return !(ip.IsLoopback() || ip.IsPrivate() || ip.IsUnspecified() || ip.IsLinkLocalUnicast() ||
			ip.IsLinkLocalMulticast() || ip.IsMulticast())
	}
	return true
}

// newAppUpdateClient checks every redirect hop with ok.
func newAppUpdateClient(rt http.RoundTripper, ok func(string) bool) *http.Client {
	return &http.Client{
		Transport: rt,
		Timeout:   appFeedTimeout,
		CheckRedirect: func(req *http.Request, via []*http.Request) error {
			if len(via) >= appFeedRedirects || !ok(req.URL.String()) {
				return errFeedURL
			}
			return nil
		},
		Jar: nil,
	}
}

// appUpdateHTTP fetches update feeds; tests replace it.
var appUpdateHTTP = newAppUpdateClient(http.DefaultTransport, feedURLOK)

// fetchFeed reads at most appFeedLimit bytes from a checked https URL.
func fetchFeed(ctx context.Context, raw string) ([]byte, error) {
	return fetchFeedWith(ctx, raw, feedURLOK)
}

// fetchFeedWith is fetchFeed with the first hop's check injected, so tests
// can use a local TLS server.
func fetchFeedWith(ctx context.Context, raw string, ok func(string) bool) ([]byte, error) {
	if !ok(raw) {
		return nil, errFeedURL
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, strings.TrimSpace(raw), nil)
	if err != nil {
		return nil, errFeedURL
	}
	req.Header.Set("User-Agent", "Cleanzo")
	resp, err := appUpdateHTTP.Do(req)
	if err != nil {
		if errors.Is(err, errFeedURL) {
			return nil, errFeedURL
		}
		return nil, err
	}
	defer resp.Body.Close()
	switch {
	case resp.StatusCode == http.StatusNotFound:
		return nil, errFeedNotFound
	case resp.StatusCode == http.StatusForbidden || resp.StatusCode == http.StatusTooManyRequests:
		return nil, errFeedRate
	case resp.StatusCode != http.StatusOK:
		return nil, fmt.Errorf("%w: status %d", errFeedBad, resp.StatusCode)
	}
	data, err := io.ReadAll(io.LimitReader(resp.Body, appFeedLimit+1))
	if err != nil {
		return nil, err
	}
	if len(data) > appFeedLimit {
		return nil, errFeedBad
	}
	return data, nil
}

// feedReason turns a fetch error into the reason code the page translates.
func feedReason(err error) string {
	switch {
	case errors.Is(err, errFeedURL):
		return "bad-url"
	case errors.Is(err, errFeedRate):
		return "rate-limited"
	case errors.Is(err, errFeedNotFound):
		return "not-found"
	case errors.Is(err, errFeedBad):
		return "bad-feed"
	default:
		return "network"
	}
}

// ---- the last check's actions --------------------------------------------------

// appUpdateCache keeps what the last check found each app's Update button
// should do. UpgradeApp only acts on what is here.
var appUpdateCache = struct {
	sync.Mutex
	m map[string]appUpdateAction
}{m: map[string]appUpdateAction{}}

func rememberAppAction(id string, a appUpdateAction) {
	appUpdateCache.Lock()
	defer appUpdateCache.Unlock()
	appUpdateCache.m[id] = a
}

func appActionFor(id string) (appUpdateAction, bool) {
	appUpdateCache.Lock()
	defer appUpdateCache.Unlock()
	a, ok := appUpdateCache.m[id]
	return a, ok
}

func resetAppActions() {
	appUpdateCache.Lock()
	defer appUpdateCache.Unlock()
	appUpdateCache.m = map[string]appUpdateAction{}
}

Rules:
- Sparkle: take the highest item without `sparkle:channel` whose `sparkle:minimumSystemVersion` does not exceed this system. Its version comes from `sparkle:version` (the element, or the `enclosure` attribute); the short version comes from `sparkle:shortVersionString`, the same way, or failing that from the `<title>` digits. Compare `version` with `CFBundleVersion` when both parse; otherwise compare short with the short version. For display, `Latest` takes the short version first.
- Electron: `provider: github` reads `https://github.com/<owner>/<repo>/releases/latest/download/<channel>-mac.yml`. `generic` reads `<url>/<channel>-mac.yml`, adding a `/` when the url lacks one. `channel` defaults to `latest`. Any other provider (`custom`, empty) gets `unknown/no-source`. A missing `version` in the yml gets `error/bad-feed`.
- Homebrew: use the cask's `version`, cleaned. `NotesURL` stays empty. The action is `brew` + token + path + latest.
- MAS: `https://itunes.apple.com/lookup?bundleId=<bid>&entity=macSoftware&country=<region>`. When `resultCount` is 0, retry with `country=us`. The action is `open-store` with `macappstore://apps.apple.com/app/id<trackId>`, and `NotesURL` is `trackViewUrl` (https only).
- Sparkle and Electron get the action `open-app` + path.
- An action is remembered only when `State == "update"`.
- `checkAppUpdates`: reset the cache first. Skip ids that `installedApp` rejects (send `unknown/no-source`). Run `brew` once, before the network work, and only if any bundle is not MAS. Run at most 6 at a time. Send each result as it finishes, then `{ID:"__done"}`. Stop on `ctx.Err()`.

type appBundle struct {

	feed               string // SUFeedURL
	mas                bool   // has an App Store receipt

// readAppBundle reads Info.plist, the App Store receipt and
// Contents/Resources/app-update.yml; false when Info.plist cannot be read.
func readAppBundle(path string) (appBundle, bool) {
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	out, err := quietCommand(exec.CommandContext(ctx, "/usr/bin/plutil", "-convert", "xml1", "-o", "-", filepath.Join(path, "Contents", "Info.plist"))).Output()
	if err != nil {
		return appBundle{}, false
	}
	m := plistValues(out)
	b := appBundle{id: path, path: path, bundleID: m["CFBundleIdentifier"], short: m["CFBundleShortVersionString"],
		build: m["CFBundleVersion"], feed: strings.TrimSpace(m["SUFeedURL"])}
	_, err = os.Stat(filepath.Join(path, "Contents", "_MASReceipt", "receipt"))
	b.mas = err == nil
	if data, err := os.ReadFile(filepath.Join(path, "Contents", "Resources", "app-update.yml")); err == nil && len(data) < 64<<10 {
		b.electron = parseYAMLScalars(data)
	}
	return b, true
}

// ---- parsers ---------------------------------------------------------------------

// parseYAMLScalars reads the top-level "key: value" lines of a small YAML
// file, unquoting single or double quotes; nested and list lines are skipped.
func parseYAMLScalars(data []byte) map[string]string {
	m := map[string]string{}
	for _, line := range strings.Split(string(data), "\n") {
		line = strings.TrimRight(line, "\r")
		if line == "" || line[0] == ' ' || line[0] == '\t' || line[0] == '-' || line[0] == '#' {
			continue
		}
		k, v, ok := strings.Cut(line, ":")
		if !ok {
			continue
		}
		v = strings.TrimSpace(v)
		if len(v) >= 2 && (v[0] == '\'' && v[len(v)-1] == '\'' || v[0] == '"' && v[len(v)-1] == '"') {
			v = v[1 : len(v)-1]
		}
		m[strings.TrimSpace(k)] = v
	}
	return m
}

	Short     string             `xml:"http://www.andymatuschak.org/xml-namespaces/sparkle shortVersionString"`
	MinOS     string             `xml:"http://www.andymatuschak.org/xml-namespaces/sparkle minimumSystemVersion"`
	Channel   string             `xml:"http://www.andymatuschak.org/xml-namespaces/sparkle channel"`
	Notes     string             `xml:"http://www.andymatuschak.org/xml-namespaces/sparkle releaseNotesLink"`
	Enclosure []appcastEnclosure `xml:"enclosure"`
}

// parseAppcast returns the newest item for this system: no channel, minimum
// system version met. version is sparkle:version, short the display version.
func parseAppcast(data []byte, osVersion string) (version, short, notes string, ok bool) {
	var feed struct {
		Items []appcastItem `xml:"channel>item"`
	}
	if xml.Unmarshal(data, &feed) != nil {
		return "", "", "", false
	}
	osParts := versionParts(cleanVersion(osVersion))
	var best []int
	for _, it := range feed.Items {
		if strings.TrimSpace(it.Channel) != "" {
			continue
		}
		if min := versionParts(cleanVersion(it.MinOS)); min != nil && osParts != nil && compareVersions(min, osParts) > 0 {
			continue
		}
		v, s := strings.TrimSpace(it.Version), strings.TrimSpace(it.Short)
		for _, e := range it.Enclosure {
			if v == "" {
				v = strings.TrimSpace(e.Version)
			}
			if s == "" {
				s = strings.TrimSpace(e.Short)
			}
		}
		if s == "" {
			if f := strings.Fields(it.Title); len(f) > 0 && versionParts(cleanVersion(f[len(f)-1])) != nil {
				s = f[len(f)-1]
			}
		}
		key := versionParts(cleanVersion(v))
		if key == nil {
			key = versionParts(cleanVersion(s))
		}
		if key == nil || (best != nil && compareVersions(key, best) <= 0) {
			continue
		}
		best, version, short, notes = key, v, s, ""
		if n := strings.TrimSpace(it.Notes); feedURLOK(n) {
			notes = n
		}
	}
	if best == nil {
		return "", "", "", false
	}
	if short == "" {
		short = version
	}
	return version, short, notes, true
}

// brewCask is one installed cask: its token, newest version and app names.
type brewCask struct {
	token, version string
	apps           []string
}

// parseBrewCasks reads `brew info --cask --installed --json=v2`. An app
// artifact is a name, optionally followed by {"target": "New Name.app"}.
func parseBrewCasks(data []byte) []brewCask {
	var info struct {
		Casks []struct {
			Token     string                       `json:"token"`
			Version   string                       `json:"version"`
			Artifacts []map[string]json.RawMessage `json:"artifacts"`
		} `json:"casks"`
	}
	if json.Unmarshal(data, &info) != nil {
		return nil
	}
	out := []brewCask{}
	for _, c := range info.Casks {
		bc := brewCask{token: c.Token, version: c.Version}
		for _, art := range c.Artifacts {
			raw, ok := art["app"]
			if !ok {
				continue
			}
			var parts []json.RawMessage
			if json.Unmarshal(raw, &parts) != nil {
				continue
			}
			for _, p := range parts {
				var name string
				var target struct {
					Target string `json:"target"`
				}
				switch {
				case json.Unmarshal(p, &name) == nil:
					bc.apps = append(bc.apps, filepath.Base(name))
				case json.Unmarshal(p, &target) == nil && target.Target != "" && len(bc.apps) > 0:
					bc.apps[len(bc.apps)-1] = filepath.Base(target.Target)
				}
			}
		}
		out = append(out, bc)
	}
	return out
}

// caskIndex maps lower-cased app bundle names to their cask.
func caskIndex(casks []brewCask) map[string]brewCask {
	m := map[string]brewCask{}
	for _, c := range casks {
		for _, a := range c.apps {
			m[strings.ToLower(a)] = c
		}
	}
	return m
}

// parseITunesLookup reads the first result of an iTunes Lookup response.
func parseITunesLookup(data []byte) (version string, trackID int64, notesURL string, ok bool) {
	var r struct {
		ResultCount int `json:"resultCount"`
		Results     []struct {
			TrackID      int64  `json:"trackId"`
			Version      string `json:"version"`
			TrackViewURL string `json:"trackViewUrl"`
		} `json:"results"`
	}
	if json.Unmarshal(data, &r) != nil || r.ResultCount == 0 || len(r.Results) == 0 || r.Results[0].Version == "" || r.Results[0].TrackID == 0 {
		return "", 0, "", false
	}
	first := r.Results[0]
	if feedURLOK(first.TrackViewURL) {
		notesURL = first.TrackViewURL
	}
	return first.Version, first.TrackID, notesURL, true
}

// installedCasks runs `brew info --cask --installed --json=v2` once; nil
// without Homebrew or when it fails.
func installedCasks(ctx context.Context) []brewCask {
	tool := findTool("brew")
	if tool == "" {
		return nil
	}
	out, err := runTool(ctx, tool, []string{"info", "--cask", "--installed", "--json=v2"}, time.Minute)
	if err != nil {
		return nil
	}
	if i := strings.IndexByte(out, '{'); i > 0 {
		out = out[i:] // runTool merges stderr; anything printed before the JSON is dropped
	}
	return parseBrewCasks([]byte(out))
}

// storeRegion is the App Store country from the system locale (zh_CN → cn).
func storeRegion() string {
	out, err := quietCommand(exec.Command("/usr/bin/defaults", "read", "-g", "AppleLocale")).Output()
	if err == nil {
		loc := strings.TrimSpace(string(out))
		if i := strings.IndexAny(loc, "@"); i >= 0 {
			loc = loc[:i]
		}
		if _, region, ok := strings.Cut(loc, "_"); ok && len(region) == 2 {
			return strings.ToLower(region)
		}
	}
	return "us"
}

// checkTarget finds an app's update source and asks it (spec §四).
func checkTarget(ctx context.Context, b appBundle, src appUpdateSources, casks map[string]brewCask) (AppUpdate, appUpdateAction) {
	u := AppUpdate{ID: b.id, Current: b.short}
	fail := func(err error) (AppUpdate, appUpdateAction) {
		u.State, u.Reason = "error", feedReason(err)
		if u.Reason == "bad-url" {
			u.State = "unknown"
		}
		return u, appUpdateAction{}
	}
	settle := func(current, latest string, a appUpdateAction) (AppUpdate, appUpdateAction) {
		u.State, u.Reason = updateState(current, latest)
		if u.State == "update" {
			u.Action = a.kind
			return u, a
		}
		return u, appUpdateAction{}
	}
	switch {
	case b.mas:
		u.Source = "mas"
		if b.bundleID == "" {
			u.State, u.Reason = "unknown", "no-source"
			return u, appUpdateAction{}
		}
		var (
			version, notes string
			id             int64
			ok             bool
		)
		for _, region := range uniq(src.region, "us") {
			q := url.Values{"bundleId": {b.bundleID}, "entity": {"macSoftware"}, "country": {region}}
			data, err := src.fetch(ctx, "https://itunes.apple.com/lookup?"+q.Encode())
			if err != nil {
				return fail(err)
			}
			if version, id, notes, ok = parseITunesLookup(data); ok {
				break
			}
		}
		if !ok {
			return fail(errFeedNotFound)
		}
		u.Latest, u.NotesURL = version, notes
		return settle(b.short, version, appUpdateAction{kind: "open-store", path: b.path, url: fmt.Sprintf("macappstore://apps.apple.com/app/id%d", id)})
	case casks[strings.ToLower(filepath.Base(b.path))].token != "":
		c := casks[strings.ToLower(filepath.Base(b.path))]
		u.Source, u.Latest = "brew", cleanVersion(c.version)
		return settle(b.short, c.version, appUpdateAction{kind: "brew", token: c.token, path: b.path, latest: cleanVersion(c.version)})
	case b.feed != "":
		u.Source = "sparkle"
		data, err := src.fetch(ctx, b.feed)
		if err != nil {
			return fail(err)
		}
		version, short, notes, ok := parseAppcast(data, src.osVersion)
		if !ok {
			return fail(errFeedBad)
		}
		u.Latest, u.NotesURL = short, notes
		a := appUpdateAction{kind: "open-app", path: b.path}
		if versionParts(cleanVersion(version)) != nil && versionParts(cleanVersion(b.build)) != nil {
			return settle(b.build, version, a)
		}
		return settle(b.short, short, a)
	case b.electron != nil:
		u.Source = "electron"
		feed := electronFeed(b.electron)
		if feed == "" {
			u.Source, u.State, u.Reason = "", "unknown", "no-source"
			if p := b.electron["provider"]; p == "github" || p == "generic" {
				u.Source, u.Reason = "electron", "bad-url"
			}
			return u, appUpdateAction{}
		}
		data, err := src.fetch(ctx, feed)
		if err != nil {
			return fail(err)
		}
		latest := parseYAMLScalars(data)["version"]
		if latest == "" {
			return fail(errFeedBad)
		}
		u.Latest = latest
		return settle(b.short, latest, appUpdateAction{kind: "open-app", path: b.path})
	}
	u.State, u.Reason = "unknown", "no-source"
	return u, appUpdateAction{}
}

// electronFeed is the electron-updater manifest URL, or "" when the
// provider is not github or generic or its address is unusable.
func electronFeed(y map[string]string) string {
	channel := y["channel"]
	if channel == "" || strings.ContainsAny(channel, "/?#") {
		channel = "latest"
	}
	switch y["provider"] {
	case "github":
		if !fileName(y["owner"]) || !fileName(y["repo"]) {
			return ""
		}
		return "https://github.com/" + url.PathEscape(y["owner"]) + "/" + url.PathEscape(y["repo"]) + "/releases/latest/download/" + channel + "-mac.yml"
	case "generic":
		base := strings.TrimSpace(y["url"])
		if !feedURLOK(base) {
			return ""
		}
		if !strings.HasSuffix(base, "/") {
			base += "/"
		}
		return base + channel + "-mac.yml"
	}
	return ""
}

func uniq(xs ...string) []string {
	var out []string
	for _, x := range xs {
		if x != "" && !slices.Contains(out, x) {
			out = append(out, x)
		}
	}
	return out
}

const appUpdateWorkers = 6

// checkAppUpdates checks ids (the page's AppInfo IDs), sending each result
// as it is known and {ID: "__done"} last. Only installed apps are read.
func checkAppUpdates(ctx context.Context, ids []string, src appUpdateSources, send func(AppUpdate) error) error {
	resetAppActions()
	if err := ctx.Err(); err != nil {
		return err
	}
	bundles := make([]appBundle, 0, len(ids))
	needBrew := false
	var sendMu sync.Mutex
	emit := func(u AppUpdate) error {
		sendMu.Lock()
		defer sendMu.Unlock()
		return send(u)
	}
	for _, id := range ids {
		b, ok := appBundle{}, false
		if installedApp(id) {
			b, ok = readAppBundle(id)
		}
		if !ok {
			if err := emit(AppUpdate{ID: id, State: "unknown", Reason: "no-source"}); err != nil {
				return err
			}
			continue
		}
		bundles = append(bundles, b)
		needBrew = needBrew || !b.mas
	}
	var casks map[string]brewCask
	if needBrew && src.brew != nil {
		casks = caskIndex(src.brew(ctx))
	}
	jobs := make(chan appBundle)
	errs := make(chan error, 1)
	var wg sync.WaitGroup
	for range appUpdateWorkers {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for b := range jobs {
				u, a := checkTarget(ctx, b, src, casks)
				if ctx.Err() != nil {
					return
				}
				if a.kind != "" {
					rememberAppAction(b.id, a)
				}
				if err := emit(u); err != nil {
					select {
					case errs <- err:
					default:
					}
					return
				}
			}
		}()
	}
feed:
	for _, b := range bundles {
		select {
		case jobs <- b:
		case <-ctx.Done():
			break feed
		case err := <-errs:
			close(jobs)
			wg.Wait()
			return err
		}
	}
	close(jobs)
	wg.Wait()
	if err := ctx.Err(); err != nil {
		return err
	}
	select {
	case err := <-errs:
		return err
	default:
	}
	return emit(AppUpdate{ID: "__done"})
}

// CheckAppUpdates streams the update state of the apps the page lists
// (ids are their AppInfo IDs), then {ID: "__done"}. Each app is checked at
// its own source: App Store, Homebrew, Sparkle or electron-updater; the
// page only asks while it is open, and closing it cancels the check.
func (Cleaner) CheckAppUpdates(ctx context.Context, ids []string, ch *mygo.Channel[AppUpdate]) error {
	return checkAppUpdates(ctx, ids, realAppUpdateSources(), ch.Send)
}

func lastLine(out string) string {
	lines := strings.Split(strings.TrimSpace(out), "\n")

			return l
		}

func cleanAppCache(path string) UninstallResult {
	res := UninstallResult{Path: path}
	if !installedApp(path) {
		res.Failed = append(res.Failed, path+": not-app")
		return res
	}

		return res

			continue
		}

// 应用页的纯逻辑：筛选、计数、版本标签、占用拆分、增长对应；不依赖 DOM / React，可单测。
import type { AppInfo, AppUpdate, GrowthApp, GrowthSummary, UninstallDetail } from "../../mygo";
import type { DictKey } from "../../ui/i18n-react";
import { growthTags, type GrowthTag } from "../process/format";

export type AppsFilter = "all" | "updates" | "idle" | "leftovers";

/** 「很久没用」：上次打开在 90 天以前，或从没打开过。 */
export const IDLE_DAYS = 90;

export function isIdle(a: AppInfo, nowSec: number): boolean {
  return a.lastOpened === 0 || nowSec - a.lastOpened > IDLE_DAYS * 86400;
}

/** 按筛选取 App；「残留」不列已装 App，由残留列表接管。 */
export function filterByKind(apps: readonly AppInfo[], f: AppsFilter, updates: ReadonlyMap<string, AppUpdate>, nowSec: number): AppInfo[] {
  if (f === "leftovers") return [];
  if (f === "updates") return apps.filter((a) => updates.get(a.id)?.state === "update");
  if (f === "idle") return apps.filter((a) => isIdle(a, nowSec));
  return [...apps];
}

export function updateCount(updates: ReadonlyMap<string, AppUpdate>): number {
  let n = 0;
  for (const u of updates.values()) if (u.state === "update") n++;
  return n;
}

/** 检查进度：done 是已有结果的 App 数，finished 是收到 __done。 */
export interface CheckProgress { done: number; total: number; finished: boolean }

export function versionTag(u: AppUpdate | undefined): string | null {

    case "mas": return "apps.src.mas";

/** 卸载路径列表拆成「App 本体 / 数据 / 缓存」；缓存只算 kind=cache，路径给「清理」用。 */

export function splitUsage(d: UninstallDetail): UsageSplit {
  const out: UsageSplit = { app: 0, data: 0, cache: 0, cachePaths: [] };
  for (const it of d.items) {
    const size = Math.max(0, it.size);
    if (it.kind === "app") out.app += size;
    else if (it.kind === "cache") { out.cache += size; out.cachePaths.push(it.path); }
    else out.data += size;
  }
  return out;
}

export interface AppRowProps {
  app: AppInfo;
  update: AppUpdate | undefined;
  growthTag: GrowthTag | null;
  open: boolean;
  /** 正在跑的动作：有就禁用这一行的按钮 */
  busy: "upgrade" | "cache" | "quit" | "uninstall" | null;
  /** 别的 brew 行正在升级：本行的 brew「更新」禁用 */
  brewBusy: boolean;
  onToggle: (id: string) => void;
  onUpgrade: (id: string) => void;
  children?: React.ReactNode;
}

/** 增长小标签文字，和进程页一样：「磁盘 7 天 +3.0 GB」。 */
export function growthTagText(tag: GrowthTag): string {
  const size = `+${human(tag.bytes)}`;
  return t(tag.window === "day" ? "process.tag.day" : "process.tag.week", { size: tag.atLeast ? t("growth.atLeast", { size }) : size });
}

/**
 * 应用页的一行：点行展开；「更新」是行旁边的独立按钮，不展开也能点。
 * 和旧的 InstalledAppRow 一样是纯函数、用全局 t（父组件用 useT 订阅语言变化），测试可以直接调用。
 */
export function AppRow({ app: a, update, growthTag, open, busy, brewBusy, onToggle, onUpgrade, children }: AppRowProps) {
  const win = isWindows();
  const ver = win ? null : versionTag(update);
  const tags: React.ReactNode[] = [];
  if (ver) tags.push(<span key="ver" className="tag ok">{ver}</span>);
  if (a.running) tags.push(<span key="run" className="tag ok">{t("apps.running")}</span>);
  if (!win && growthTag && tags.length < 2) tags.push(<span key="grow" className="tag grow">{growthTagText(growthTag)}</span>);
  const canUpgrade = !win && update?.state === "update" && update.action !== "";
  const upgradeBlocked = busy !== null || (update?.action === "brew" && (brewBusy || a.running));
  return (
    <div className="app-item card" data-key={a.id}>
      <div className="app-head">
        <button type="button" className="app-row" aria-expanded={open} onClick={() => onToggle(a.id)}>
          <AppIcon path={a.path} box="app-icon">
            <div className="app-icon">{a.name.charAt(0).toUpperCase()}</div>
          </AppIcon>
          <div className="app-main">
            <div className="app-name">
              {a.name}
              {tags.slice(0, 2)}
              {!ver && a.version && <span className="ver muted"> {a.version}</span>}
            </div>
            <div className="app-path muted" title={win ? a.publisher : a.path}>{win ? a.publisher : a.path}</div>
          </div>
          <div className="app-meta">
            <div className="app-size">{human(a.size)}</div>
            {(!win || a.lastOpened !== 0) && <div className="app-used muted small">{a.lastOpened ? fmtDate(a.lastOpened) : t("apps.never")}</div>}
          </div>
          <ChevronRight className="app-chev h-4 w-4" aria-hidden="true" />
        </button>
        {canUpgrade && (
          <Button size="sm" className="app-upgrade" disabled={upgradeBlocked} onClick={() => onUpgrade(a.id)}>
            {busy === "upgrade" ? t("apps.updating") : t("apps.update")}
          </Button>
        )}
      </div>
      {children}
    </div>
  );
}

/** 带离场动画的行（列表里用这个）；动效 hook 单独挂载，AppRow 本身仍可直接调用。 */
export function AnimatedAppRow(props: AppRowProps & { leaving?: boolean }) {
  const ref = React.useRef<HTMLDivElement>(null);
  useLeaveAnimation(ref, !!props.leaving);
  return React.cloneElement(AppRow(props), { ref, "aria-hidden": props.leaving || undefined, inert: props.leaving || undefined } as never);
}

export function checkStep(
  prev: { active: boolean; loaded: boolean; running: boolean; finished: boolean },
  next: { active: boolean; loaded: boolean },
): "start" | "stop" | "none" {
  if (!next.active || !next.loaded) return prev.running ? "stop" : "none";
  if (prev.running || prev.finished) return "none";
  return "start";
}
